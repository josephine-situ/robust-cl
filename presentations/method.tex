\begin{frame}{Formulation}
  \textbf{Nominal}:
  \[
    \min_{x\in\mathcal{X}}\; c^\top x \quad\text{s.t.}\quad \hat f(x;\theta^\ast)\le b .
  \]
  \begin{itemize}
    \item Noisy labels $\Rightarrow$ different $\theta^\ast(\delta)$ $\Rightarrow$ infeasible
      decisions.
    \item \textbf{Robust}: feasible against \emph{every} reachable model.
  \end{itemize}
  \[
    \min_{x\in\mathcal{X}} c^\top x
    \;\;\text{s.t.}\;\; \max_{\delta\in\mathcal{D}} f(x;\theta^\ast(\delta))\le b,
  \]
  \[
    \theta^\ast(\delta)=\arg\min_{\theta\in\Theta_{\text{feas}}}\mathcal L(\theta;X,y+\delta)
  \]
\end{frame}

% ===========================================================================
\begin{frame}{Summary}
  \begin{enumerate}
    \item Select model class (type + hyperparameters).
    \item For each method:
      \begin{enumerate}
        \item Calibrate one robustness parameter ($\rho$) per method on held-out
              folds. Ablate other method-specific parameters.
        \item Evaluate fixed parameters on repeated subsampled training draws,
              against a fixed ground truth.
      \end{enumerate}
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Trees enter the MIP exactly\implnote}
  \begin{itemize}
    \item Leaf boxes separated by a $\le10^{-5}$ band on \emph{both} sides of each
      split, clamped so no training row loses its leaf. Verified
      $\le3\times10^{-7}$ vs the trained model, over all rows and perturbed refits.
    \item Both sides needed: a bound left at $\theta$ is still met at
      $\theta-$\texttt{FeasibilityTol}, routing $x$ to the wrong leaf.
    \item Branches routed in \texttt{float32} --- both libraries traverse there,
      so $x$ within an ulp of $\theta$ goes the way the model sends it.
    \item \texttt{IntFeasTol}$=10^{-9}$: big-M turns integrality slack into
      $M\!\cdot\!$\texttt{IntFeasTol} of $x$-slack ($M\!\approx\!46$). At the
      $10^{-5}$ default a free-$x$ adversary extracts $1.6\times10^{-2}$.
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: synthetic}
  True constraint, known exactly (evaluation only):
  \[
    f_{\text{true}}(x) = \sum_{j=1}^d x_j^2 + 0.5\prod_{j=1}^d x_j, \qquad x\in[0,1]^d
  \]
  Noisy labels $y_i = f_{\text{true}}(x_i)+\varepsilon_i$,
  $\varepsilon_i\sim\mathcal N(0,\sigma^2)$:
  \[
    \min_{x\in[0,1]^d} \; -\sum_{j=1}^d x_j \quad\text{s.t.}\quad \hat f(x;\theta)\le b,
    \qquad b=0.5\,d
  \]
  \begin{itemize}
    \item One LP, one learned constraint, no context.
    \item $n=2500$, $d=2$, $\sigma=0.1$. Coherence vacuous (one outcome).
    \item Embedded model \textbf{CV-selected} (MLP $(50)$, CV $R^2$ 0.964); was a
      hard-coded \texttt{rf}. $n$ raised from 200: there the CV winner amplified a
      $0.128$/row label shift into a $0.62$ refit spread and the wrapper
      prescribed the box corner.
    \item \textbf{Read feasibility here with care.} The tuning judge is a fitted
      proxy (7-model ensemble, incl.\ MLP), and a constrained optimum sits
      \emph{on} the boundary --- error sd $0.033$, $26\%$ of verdicts flipped vs
      $f_{\text{true}}$ within $|f-b|<0.05$. The MLP member helps ($0.038$, $28\%$
      without) but does not fix it. Not neutral between methods: robust methods
      leave slack, nominal does not.
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: reactor (C-MICL)}
  \small
  DMA-MR membrane reactor, methane $\to$ hydrogen + benzene (Ovalle et al.\ 2025
  App.~D.1). $\hat h$ predicts outlet $F_{\mathrm{C_6H_6}}$ [mg/h]; truth = the
  ODE overleaf.
  \vspace{-0.3em}
  \[
    \begin{aligned}
      \min_{x}\;\; & c^\top x, \quad x=(v_0,\,v_{\mathrm{He}},\,T,\,d_t,\,L),
        \quad c=\mathbf 1 \\
      \text{s.t.}\;\; & \hat h(x;\theta)\ \ge\ 50 && \text{(D.1g --- \emph{learned})}\\
      & 10 \le L/d_t \le 150, \qquad 20 \le v_0/L \le 120 && \text{(D.1b--e,}\\
      & 0.75 \le v_0/v_{\mathrm{He}} \le 3, \qquad v_0 \le 1.1\,T
        && \text{cross-multiplied}\\
      & \ell \le x \le u && \text{$\to$ 7 linear ineqs.)}
    \end{aligned}
  \]
  \vspace{-1.1em}
  \begin{center}\footnotesize
  \begin{tabular}{@{}lccccc@{}}
    \toprule
      & $v_0$ [cm$^3$/h] & $v_{\mathrm{He}}$ [cm$^3$/h] & $T$ [K] & $d_t$ [cm] & $L$ [cm] \\
    \midrule
    $\ell$ & 450 & 450 & 997.18 & 0.5 & 10 \\
    $u$    & 1500 & 1500 & 1348.12 & 2.0 & 100 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{-0.5em}
  \begin{itemize}
    \item $n=1000$ designs uniform over the box; labels $+\,\mathcal N(0,2^2)$,
      range $11$--$70$. Single decision, no context.
    \item Lower bound $\Rightarrow$ $w=-1$, $\text{rhs}=-50$ in the repo's
      $\sum_m w_mf_m\le\text{rhs}$ form.
    \item $c$ \textbf{fixed} at $1/\mathrm{span}_i$ --- a new $c$ is a different
      problem. Raw units are incommensurate ($v_0,v_{\mathrm{He}}$ span $1050$ vs
      $d_t$'s $1.5$), so under ones $d_t$ carried $0.06\%$ of the objective range;
      balanced gives each variable $20\%$. \emph{Not} C-MICL's per-instance draw
      --- theirs is an average over $100$ cost vectors.
  \end{itemize}
\end{frame}

\begin{frame}{Reactor: the ground truth is an ODE system}
  \small
  Carrasco \& Lima (2017), integrated along the tube, $z:0\to L$. State
  $F(z)\in\mathbb R^{8}$: tube-side $(\mathrm{CH_4},\mathrm{C_2H_6},\mathrm{H_2},
  \mathrm{C_6H_6})$ and the same four on the shell (sweep) side. Concentrations
  $C_i=F^t_i/v$ with $v=v_0F^t_{\text{tot}}/F_{t0}$; cross-section
  $A_t=\tfrac\pi4 d_t^2$, catalyst factor $\kappa=(1-\nu_b)\eta=0.45$. Two
  equilibrium-limited reactions, as the vendored RHS writes them (self-consistent,
  $\mathrm H$ unbalanced):
  $2\mathrm{CH_4}\rightleftharpoons\mathrm{C_2H_6}+2\mathrm{H_2}$,
  $3\mathrm{C_2H_6}\rightleftharpoons\mathrm{C_6H_6}+3\mathrm{H_2}$.
  \vspace{-0.4em}
  \[
    r_1 = 3600\,k_1C_{\mathrm{CH_4}}\Big(1-\tfrac{k_1^{-}C_{\mathrm{C_2H_6}}C_{\mathrm{H_2}}^{2}}
      {k_1C_{\mathrm{CH_4}}^{2}}\Big),
    \quad
    r_2 = 3600\,k_2C_{\mathrm{C_2H_6}}\Big(1-\tfrac{k_2^{-}C_{\mathrm{C_6H_6}}C_{\mathrm{H_2}}^{3}}
      {k_2C_{\mathrm{C_2H_6}}^{3}}\Big)
  \]
  \vspace{-0.6em}
  \[
    \begin{aligned}
      \tfrac{d}{dz}F^t_{\mathrm{CH_4}} &= -\kappa r_1A_t - J_{\mathrm{CH_4}}, &\quad
      \tfrac{d}{dz}F^t_{\mathrm{H_2}} &= \kappa(r_1+r_2)A_t - J_{\mathrm{H_2}}, \\[0.15em]
      \tfrac{d}{dz}F^t_{\mathrm{C_2H_6}} &= \kappa(\tfrac12 r_1-r_2)A_t - J_{\mathrm{C_2H_6}}, &\quad
      \tfrac{d}{dz}F^t_{\mathrm{C_6H_6}} &= \tfrac13\kappa r_2A_t - J_{\mathrm{C_6H_6}}, \\[0.15em]
      \tfrac{d}{dz}F^s_i &= +J_i, &\quad
      J_i &= Q_i\,\pi d_t\big(p_{i,t}^{1/4}-p_{i,s}^{1/4}\big)
    \end{aligned}
  \]
  \vspace{-0.7em}
  \begin{itemize}\footnotesize
    \item $J_i$ leaves the tube for the shell ($p$ in atm; $Q_i=Q$ for
      $\mathrm{H_2}$, $Q/S$ otherwise, $S=1500$): $\mathrm{H_2}$ leaves
      $1500\times$ faster, so Le Chatelier drives both equilibria forward --- the
      point of the membrane.
    \item Inlet pure methane $F_{t0}=P_tv_0/(RT)$; sweep helium
      $F_{\mathrm{He}}=P_sv_{\mathrm{He}}/(RT)$ enters the shell \emph{total}
      only. Oracle $h(x)=10^{3}M_B\,F^t_{\mathrm{C_6H_6}}(L)$.
    \item RHS vendored unchanged from \texttt{opyrability} (the code C-MICL cite).
      \texttt{solve\_ivp} at rtol$=$atol$=10^{-10}$: the defaults are $3.8$ off on
      a Table~1 row.
  \end{itemize}
\end{frame}

\begin{frame}{Reactor: why this instance}
  \small
  \begin{itemize}
    \item \textbf{Ground truth is a system of ODEs}, not a fitted model --- the only
      such oracle here, and the reason for the instance. Feasibility is
      \emph{integrated}, so the judge carries no error of its own at the boundary,
      which is exactly where a constrained optimum sits.
    \item \textbf{Two judges, kept apart.} ODE $\Rightarrow$ final evaluation and
      audits. Proxy ensemble (7 types, incl.\ MLP) $\Rightarrow$ $\rho$ tuning:
      calibrating $\rho$ against exact truth is the same error as pinning
      $\mathcal D$ to synthetic's known $\sigma$.
    \item \textbf{Design $\ne$ feasible set}: the $1000$ rows are uniform over the
      box, but the optimizer lives in the box $\cap$ the ratio constraints. Only
      $216/1000$ ($21.6\%$) rows meet the ratios, $30$ ($3.0\%$) also reach $50$.
      Kept --- it is what the paper specifies, and it makes boundary uncertainty
      genuine.
    \item \textbf{Published baseline}: their Fig.~1 has W-MICL --- our wrapper ---
      at $0.45$--$0.85$ ground-truth feasibility, never reaching $0.9$.
    \item Validated: RF $0.855$ vs their $0.864$, GBM $0.912$ vs $0.907$
      (5-fold $R^2$). Table 1 agrees to a consistent $-1.6\%$.
  \end{itemize}
\end{frame}

\begin{frame}{Robust methods summary}
  \small
  {\footnotesize\mc{Red} $=$ beyond the basic setting: needed only for multiple
  constraints / contexts / a learned objective (gastric).}
  \[
    \mathcal D_c = \{\delta:\ \lVert\delta\rVert_2\le R_c\},
    \qquad R_c = \rho\,\mathrm{scale}(y_c)\,\sqrt n
  \]
  \begin{itemize}
    \item \mc{Each $y$ has its own $\mathcal D_c$ (radius $R_c$), but the \emph{same} $\rho$.}
    \item Each method has its own dials; we tune only $\rho$, for comparability.
    \item CP/wrapper embed scenarios as multiple \emph{constraints};
      multiple \emph{directed} scenarios $\Rightarrow$ infeasible MIP.
    \item \textbf{Two face no $\mathcal D$}: tuned nominal ($m$) and C-MICL
      ($\alpha$). Flat in $\rho$ by construction $\Rightarrow$ on the $\rho$ axis
      they are scored once, as reference levels; on the \emph{primary} axis each
      walks its own dial like the rest.
  \end{itemize}
  \begin{center}\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    Method & Worst case found by & Scenario becomes \\
    \midrule
    Robust Reg. & \textbf{directed}: $\delta = R\,r/\lVert r\rVert$ & the \emph{fit} \\
    CP          & \textbf{random} draws, cut the argmax  & a \emph{cut} \\
    Wrapper     & \textbf{random} draws, chance constraint & $P$ \emph{constraints} \\
    C-MICL      & \textbf{no $\mathcal D$}: held-out residuals & a \emph{tightening} $\hat q\hat u(x)$ \\
    Margin      & \textbf{no $\mathcal D$}: nothing --- $m$ is fitted & a shift of $b$ \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{$\rho$: one size parameter, swept not tuned}
  \small
  \begin{itemize}
    \item \textbf{One parameter}, not two like a budget uncertainty set.
    \item $\sqrt n$: $\rho=1$ = L2 norm of iid noise at one
      unexplained sd/row. Transfers across $n{=}200$ and $n{=}320$.
    \item \textbf{Derived}: $\rho^\ast(\text{method})$ at a fixed feasibility target.
  \end{itemize}
  \medskip
  \textbf{The sweep is per-method \emph{parameter}}, each method walked along the
  dial that actually makes it conservative:
  \vspace{-0.3em}
  \begin{center}\footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    Swept & Methods & A point on the axis is \\
    \midrule
    $\rho$ & CP, wrapper, robust\_reg & an \emph{assumed} radius \\
    $m$    & tuned nominal            & a \emph{tightening} that is paid for \\
    ---    & nominal, C-MICL          & nothing: scored once, a level \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{-0.2em}
  \begin{itemize}\footnotesize
    \item One grid: all are in unexplained sds. \textbf{Within a grid value the
      three $\rho$ methods still face the \emph{same} $\mathcal D$} --- that is
      what the shared-$\mathcal D$ comparison rests on.
    \item $\rho^\ast$ is \emph{capacity}, $m^\ast$ is \emph{price}: not comparable
      as numbers. Compare \textbf{objective at equal feasibility} --- the curve.
  \end{itemize}
\end{frame}

\begin{frame}{The scenario bank --- $\mathcal D$, made finite\implnote}
  \small
  $B$ draws from $\mathcal D$; one model retrained per draw \mc{per outcome}.
  \begin{itemize}
    \item Draw $b$ is a \textbf{pure function of $(\texttt{seed},b)$}
      $\Rightarrow$ a bank of size $P<B$ is a genuine \emph{prefix} of one of size $B$.
    \item \texttt{random\_state} fixed across members: the scenario is the
      \emph{only} source of variation.
    \item \mc{\textbf{Incoherent by default}: independent draws per outcome.
      Coherent (the ablation): one standardized direction shared across the
      \emph{group}, scaled by each outcome's own radius.}
    \item \mc{\textbf{DLT is derived, not drawn}: perturb the 4 components,
      un-percentile, apply $1-\prod(1-\cdot)$, re-percentile. Reaches the bank
      (CP, wrapper); robust reg's per-outcome adversary is unlinked.}
    \item \mc{Percentile labels are clipped to $[0,1]$ (clipping only shrinks
      $|\delta_i|$, so the draw stays in $\mathcal D$). \textbf{Every} method that
      faces $\mathcal D$ clips --- bank draws \emph{and} robust reg's training
      adversary --- so the shared set is $\mathcal D\cap[0,1]$. It binds: at
      $\rho{=}1$, 45--49\% of shifted labels leave $[0,1]$ and the realizable
      shift roughly \textbf{halves} ($\lVert\delta\rVert$ 4.56$\to$2.56, DLT).}
    \item \textbf{Built once}, not per iteration: with the draws fixed,
      $\theta(\delta_b)$ no longer depends on $x^\ast$, so CP's
      worst-violation-over-the-bank is monotone and $\tau$ means something.
  \end{itemize}
\end{frame}

\begin{frame}{\mc{Why incoherent, and why DLT is derived (gastric)}}
  \small
  Coherent draws assert residual correlation $+1$ across outcomes;
  incoherent assert $0$. OOF residuals ($n=145$, forward-chaining):
  \begin{center}\small
  \begin{tabular}{@{}lcc@{}}
    \toprule
    & percentile labels & raw labels \\
    \midrule
    Toxicity pairs (excl.\ DLT) & $+0.28$ & $+0.22$ \\
    OS vs.\ every toxicity      & $+0.06$ & $+0.02$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item \textbf{OS is uncoupled}. \texttt{coherent\_exclude} keeps it off the
      shared direction whenever coherence is on --- a \emph{grouping}, not a flag.
    \item DLT excluded \emph{from the average}: $\text{DLT}=1-\prod_c(1-y_c)$
      holds exactly (2e-16), so its $+0.44$--$0.80$ row is construction.
    \item \textbf{So DLT is not drawn at all.} Its $\delta$ is
      $\text{derive}(\text{perturbed components})-\text{baseline}$: identity error
      $\le 6.25$e-3 (a \texttt{percentileofscore} tie, $2/320$ of a rank) against
      $0.43$--$0.99$ for a free draw. Its size is now an \emph{output} ---
      $1.19\times$ its own $R_{\text{DLT}}$ at $\rho{=}0.25$, $0.89\times$ at
      $\rho{=}1$ --- so that radius is \textbf{reported, not imposed}.
    \item \textbf{$\Rightarrow$ incoherent is the default.} The one block that
      genuinely was $\approx+1$ is gone; what a shared direction would still model
      is the $+0.28$ block, and $0$ is the closer of the two assertions available.
    \item Still \emph{asserted}: we impose $0$ where truth is $+0.28$. Coherent
      survives as the ablation --- and is \emph{required} for the
      $\alpha{=}0\equiv\tau\to0$ check: CP's separation follows the bank, so
      \texttt{--coherent} is the arm where that equivalence is defined at all.
    \item \textbf{Product-set coverage}: on the incoherent path each constraint
      ranks the bank itself, so separation searches $B$ points \emph{per
      constraint} rather than $B$ joint points --- the objection incoherence
      raised, answered on CP's side. The wrapper is unchanged.
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression --- robustify the \emph{fit}}
  \[
    \min_\theta \max_{\delta\in\mathcal D_c} \mathcal L(\theta; X, y_c+\delta)
  \]
  \begin{itemize}
    \item \textbf{Linear, unbounded labels}: exact SOCP, solved once (next 2 slides).
    \item \textbf{Trees / ensembles}: no closed form --- alternate worst $\delta$ /
      retrain, $K=5$ times.
    \item \mc{\textbf{Bounded labels}: same alternation, each $\delta$ clipped ---
      $\max\lVert r+\delta\rVert^2$ over a ball $\cap$ box is not
      $(\lVert r\rVert+R)^2$. Costs GI its closed form, buys the shared
      $\mathcal D$.}
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression: the linear counterpart (1/2)\implnote}
  \small
  The penalty does not see $\delta$; the perturbed residual is the nominal one shifted.
  With $P_\alpha(\beta)=\alpha\lambda\lVert\beta\rVert_1+\tfrac{\alpha}{2}(1-\lambda)\lVert\beta\rVert_2^2$
  (sklearn's $\alpha,\lambda$ --- \emph{not} the wrapper's):
  \[
    \mathcal L(\beta,b_0;\,y+\delta)
      = \tfrac{1}{2n}\big\lVert \underbrace{(y-Z\beta-b_0\mathbf 1)}_{r}+\delta\big\rVert_2^2
        + P_\alpha(\beta).
  \]
  On the ball the inner max is closed form --- push every residual outward along
  itself, $\delta^\ast = R\,r/\lVert r\rVert$:
  \[
    \max_{\lVert\delta\rVert_2\le R}\mathcal L
      = \tfrac{1}{2n}\max_{\lVert\delta\rVert_2\le R}\lVert r+\delta\rVert_2^2 + P_\alpha(\beta)
      = \tfrac{1}{2n}\big(\lVert r\rVert_2+R\big)^2 + P_\alpha(\beta).
  \]
  \begin{itemize}
    \item $\delta$ is \textbf{gone}: the outer $\min_\theta$ is an ordinary fit in
      $(\beta,b_0)$. The $\beta$ terms are the nominal ElasticNet's, untouched ---
      only the data term moved, $\lVert r\rVert^2\to(\lVert r\rVert+R)^2$.
    \item Still convex: $r$ affine in $(\beta,b_0)\Rightarrow\lVert r\rVert_2$ convex;
      $s\mapsto(s+R)^2$ convex and increasing on $s\ge0$.
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression: \dots is an exact SOCP (2/2)\implnote}
  \small
  Epigraph $\nu \ge \lVert r\rVert_2$ (one second-order cone), expand
  $(\nu+R)^2=\nu^2+2R\nu+R^2$:
  \[
    \min_{\beta,b_0,r,\nu}\;\; \tfrac{1}{2n}\big[\nu^2 + 2R\,\nu + R^2\big] + P_\alpha(\beta)
    \quad\text{s.t.}\quad
    \begin{cases}
      r = y - Z\beta - b_0\mathbf 1,\\[0.2em]
      \lVert r\rVert_2^2 \le \nu^2,\;\; \nu\ge0 .
    \end{cases}
  \]
  \begin{itemize}
    \item Relaxation is \textbf{tight}: $\nu$ appears nowhere else, and the objective
      \emph{increases} in $\nu\ge0$. Slack $\nu>\lVert r\rVert$ is charged
      $\tfrac{1}{2n}(\nu^2+2R\nu)$ and buys nothing $\Rightarrow \nu^\ast=\lVert r^\ast\rVert_2$.
    \item Constant $R^2$ kept --- puts the objective on the box arm's scale.
    \item \textbf{Simpler than the box$\cap$L1 case}, which needed the top-$m$ sum
      $S_m(|r|)$ and its $2n{+}1$ LP epigraph variables. Here: one cone.
    \item Convex QP $\to$ convex SOCP. Penalty still matches sklearn's ElasticNet
      scaling $\Rightarrow$ comparable to the nominal fit; $R=0$ recovers it.
    \item Exact for the \textbf{raw} ball only: $\delta^\ast=R\,r/\lVert r\rVert$ is
      unrealizable once labels are bounded, so \mc{gastric GI} takes the loop.
  \end{itemize}
\end{frame}

\begin{frame}{What $R$ actually does to the linear arm: it de-regularizes}
  \small
  $\nu^2$ \emph{is} the nominal SSE and $R^2$ is a constant, so $R$ reaches $\beta$
  only through $2R\nu = 2R\lVert r\rVert_2$. Stationarity:
  \[
    -\tfrac{1}{n}\Big(1+\tfrac{R}{\lVert r\rVert}\Big)Z^\top r + \partial P_\alpha(\beta) \ni 0
    \;\;\overset{\times\,\kappa}{\Longleftrightarrow}\;\;
    -\tfrac{1}{n}Z^\top r + \partial P_{\kappa\alpha}(\beta) \ni 0 ,
  \]
  where $\kappa=\lVert r\rVert_2/(\lVert r\rVert_2+R)\in(0,1]$.
  \begin{itemize}
    \item So the ball's robust fit \textbf{is} a nominal ElasticNet at
      $\alpha_{\text{eff}}=\kappa\alpha$ --- verified to $5\times10^{-4}$
      ($n{=}200$, $p{=}8$, $\alpha{=}0.1$): $\kappa=0.92,\,0.85,\,0.53$ at
      $\rho=0.1,\,0.2,\,1.0$; $\lVert\beta\rVert$ \emph{grows} $2.547\to2.676$.
    \item $\kappa\approx 1/(1+\rho)$ when in-sample residual sd $\approx$ \texttt{oof\_sd}.
    \item \textbf{At $\alpha=0$, $R$ does nothing}: $\arg\min(\lVert r\rVert+R)^2=\arg\min\lVert r\rVert^2$
      = OLS. All of this arm's ``robustness'' is mediated by the penalty.
    \item Direction is \emph{away} from shrinkage --- the label ball's power does not
      depend on $\beta$ (unlike feature perturbations, which give
      $\lVert r\rVert+R\lVert\beta\rVert$, i.e.\ lasso/ridge).
    \item Gastric linear outcomes: GI ($\alpha{=}0.01$) and OS ($\alpha{=}0.1$), both
      pure lasso. $\rho{=}1$ roughly halves their penalty.
  \end{itemize}
\end{frame}

\begin{frame}{Fit-robust vs.\ decision-feasible: where the $\max$ sits}
  \small
  Both solve $\min_{x\in\mathcal X} c^\top x$ over the \emph{same} $\mathcal D$.
  \textbf{robust\_reg} --- robustify the \emph{fit}:
  \[
  \begin{aligned}
    \theta^\ast &= \arg\min_\theta
      \underbrace{\max_{\delta\in\mathcal D}\mathcal L(\theta;\,y+\delta)}
                 _{\text{\scriptsize max on the \textsc{loss}, no $x$}} \\
    \text{s.t.}\;\; & \underbrace{f(x;\theta^\ast)\le b}
                 _{\text{\scriptsize a \textsc{nominal} constraint}}
  \end{aligned}
  \]
  \textbf{CP / wrapper} --- robustify the \emph{decision}:
  \[
  \begin{aligned}
    \theta(\delta) &= \arg\min_\theta
      \underbrace{\mathcal L(\theta;\,y+\delta)}
                 _{\text{\scriptsize plain fit, \textsc{no} max}} \\
    \text{s.t.}\;\; & \underbrace{\max_{\delta\in\mathcal D} f(x;\theta(\delta))\le b}
                 _{\text{\scriptsize max on the \textsc{constraint}}}
  \end{aligned}
  \]
  \begin{itemize}
    \item An $x$-free max cannot give an $x$-dependent answer.
    \item Top collapses $\mathcal D$ to \textbf{one} $\theta^\ast$ before the
      optimizer sees anything --- fragility per decision is lost.
    \item Bottom \textbf{is} the \textbf{Formulation} slide; robust\_reg departs from it.
  \end{itemize}
\end{frame}

\begin{frame}{Wrapper --- chance-constrain a sample of $\mathcal D$}
  \begin{itemize}
    \item $P=20$ draws from the \emph{shared bank} (the sphere $\lVert\delta\rVert_2=R_c$, not
      bootstrap resamples); $P$ replicates of each learned model, \textbf{all embedded}.
    \begin{itemize}
      \item Differs from bootstrap resamples (prop 0.5) used by Maragno et al. for WFP example as
        it is smoother at small n. Also aligns with other robust methods.
    \end{itemize}
    \item \mc{Incoherent (default): $z_{c,p}$ per constraint. Coherent (ablation):
      $z_p=1$ iff $x$ satisfies \emph{every} constraint under replicate $p$ --- one
      indicator per draw, and the arm CP's shared-$b$ cut has an analogue for.}
      \vspace{-0.2em}
      {\scriptsize
      \[
        \Delta_p=\left(
        \begin{array}{r|rrrc}
                       & \text{arm}_1 & \text{arm}_2 & \text{arm}_3 & \cdots\\\hline
          \text{DLT}   & +0.31 & -0.22 & +0.05 & \cdots\\
          \vdots       & \vdots & \vdots & \vdots & \\
          \text{GI}    & +0.33 & -0.23 & +0.05 & \cdots\\\hline
          \text{OS}    & -1.8\phantom{0} & -0.9\phantom{0} & +2.4\phantom{0} & \cdots
        \end{array}\right)
      \]}
      \vspace{-0.5em}
      \begin{itemize}\scriptsize
        \item \mc{Shown: the \textbf{coherent} ablation --- one $u_p$ scaled by
          each $R_c$, OS on its own. \textbf{Default}: every row its own
          direction, the DLT row \emph{derived} from the other four.}
      \end{itemize}
      \vspace{-0.3em}
    \item Chance constraint: $\tfrac1P\sum_p z_p \ge 1-\alpha$.
    \item \mc{OS robustification \textbf{off} by default --- matches CP.}
    \item Parameter: $\alpha$. $P$ is capped by MIP size.
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes (ours)}
  \small
  \begin{enumerate}
    \item Solve master $\to x^\ast$ \mc{--- gastric: one \emph{shared} master, solved
      per \textbf{anchor} (10 k-medoids over training contexts; every test point
      reuses it)}.
    \item \textbf{Separate}: score bank draws at $x^\ast$; cut the worst if its
      distance $>$ tolerance.
    \item \textbf{Feasibility rollback}: add the cut \emph{tentatively}; if the master
      then has no solution \mc{at any \textbf{protected anchor}}, remove it,
      \textbf{permanently reject} it, try the next candidate. Repeat.
  \end{enumerate}
  \mc{Two separation paths, \textbf{auto-selected from the bank} --- the adversary
  matches the set it is drawn from, so \texttt{--coherent} flips draws \emph{and}
  adversary as one ablation.}
  \begin{itemize}
    \item \mc{\textbf{Incoherent} (gastric, production): $\mathcal D$ is the
      \emph{product} set, so which relabelings one draw pairs is a sampler
      artefact. The draws are ranked \textbf{per constraint} and one model is
      admitted for \emph{each} of the 5 $\Rightarrow$ up to 5 cuts/iteration from
      up to 5 different draws. $\tau$ met \textbf{per constraint} (mean exceedance
      over anchors); CP stops only when \emph{every} constraint is within it.}
    \item \mc{\textbf{Forward} rollback: constraints most-violating first; a model
      that breaks a protected anchor is rolled back and that constraint tries its
      \emph{next} most-violating model --- so an iteration ends with a model added
      for each constraint. Only that attempt's models are removed, never the
      constraint's other cuts. (\texttt{peel}: stage all, drop from the
      least-violating end, no fallback --- cheaper, heuristic attribution.)}
    \item \mc{\textbf{OS robustification off}: ablation gave worse feasibility \emph{and} OS.}
    \item $\tau$ (stopping tolerance, in unexplained SD) is CP's \emph{only}
      parameter --- no coverage dial.
    \item Separation \emph{pool}: $B{=}200$ scored; wrapper embeds all $P{=}20$.
    \item \mc{\textbf{Coherent} path (the \texttt{--coherent} ablation,
      \emph{unchanged}): draws share a direction, so one draw \emph{is} one
      relabeling --- cut it whole, ranked by mean exceedance over anchor $\times$
      outcome. Where the $\alpha{=}0\equiv\tau\to0$ check lives.}
    \item \mc{$\tau$ is \emph{not} comparable across the paths: coherent averages
      over the 5 outcomes, incoherent does not, so incoherent distances run
      ${\approx}5\times$ larger (gastric, same bank: $0.062$ vs $0.244$).}
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes: what makes it terminate\implnote}
  \small
  Each of these was found by a run, not by reading. Together: gastric went from an
  exact period-4 cycle to \texttt{status=optimal} in 19 iterations.
  \begin{enumerate}
    \item \textbf{Fixed bank} (\texttt{scenario\_source: noise}). Redrawing each
      iteration compared iteration-$k$'s sample against iteration-0's $d_0$.
    \item \textbf{MIP gap $10^{-4}$} in the cut loop (was 1\%). Gastric obj
      ${\approx}10$ $\Rightarrow$ gap ${\approx}0.1$ vs distances ${\approx}0.007$:
      cuts below the solver's tolerance left $x^\ast$ unmoved. Synthetic never hit
      this (obj ${\approx}1.2$, distances ${\approx}0.1$). \mc{Now
      \texttt{optimization.mip\_gap}: one value for \emph{every} method and for the
      prescribe-time re-solve --- the comparison is on the objective, so a
      per-method gap confounds it.}
    \item \textbf{Nothing is removed} from the master: pruning off,
      \texttt{cut\_eviction: reject}. Removing a cut lets a previous $x^\ast$ recur
      --- that is what a cycle is. Eligible set strictly shrinks
      $\Rightarrow$ terminates in $\le B$.
    \item \mc{\textbf{Protected anchor set is fixed}, not recomputed: the anchors the
      \emph{nominal} fit could serve (8/10 on gastric), tested \textbf{set-wise}.}
    \item \mc{\textbf{Rejections cached} --- sound given (3)+(4). Without it, rollbacks
      grew $8\to44$ per iteration.}
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\mc{Cutting Planes: coverage (anchors)}\implnote}
  \begin{itemize}
    \item Adding cuts can make the master infeasible at some anchor.
    \item A cut breaking the protected set is \textbf{rolled back and permanently
      rejected}. So $\tau$ is CP's only robustness parameter. \mc{On the
      incoherent path rejection is keyed \textbf{(constraint, draw)} --- a draw cut
      for one constraint stays eligible for the others --- which bounds the whole
      run at $C\times B$ rollbacks.}
    \item The protected set is \textbf{fixed}, not recomputed per iteration ---
      the anchors the \emph{nominal} fit could serve (8/10 on gastric), tested
      set-wise so CP cannot trade one patient's feasibility for another's.
    \item Rejections are \textbf{cached} (113/200 draws on gastric); a fixed
      protected set plus a monotone master makes that sound.
    \item \mc{The cap is a count budget, $\lfloor\alpha_{\text{cp}}n\rfloor$ less
      the anchors nominal already failed; \textbf{pinned at $0$} for every result
      and moved only by its \emph{ablation}. Above $0$ CP does trade one patient
      for another, and which one is an artefact of the anchor order.}
  \end{itemize}
\end{frame}

\begin{frame}{Calibration: primary axis is each method's own dial}
  Held-out CV, after model selection. $\rho$ \textbf{fixed} at two columns
  (gastric $0.5,1$; reactor $2,3$, and $5,6$ for the wrapper); each walks
  \emph{its own} dial.
  \begin{itemize}
    \item \textbf{A method may take its OWN columns.} The wrapper on the
      reactor is out of \emph{dial} at $\alpha{=}0$ on $\{2,3\}$ (all $P{=}20$
      models must hold; only $0.40$), and first clears $0.9$ at $\rho{=}6$. Not
      a confound --- the comparison is at equal \textbf{feasibility}, not equal
      $\rho$, so the extra capacity is priced by the curve; reporting it on a
      column where it has no $\text{dial}^\ast$ would measure the column.
    \item Reported object: the curve in \textbf{objective $\times$ feasibility}.
      ``At equal feasibility, whose decisions are better'' is a two-axis claim.
    \item Dials: CP $\tau$, wrapper $\alpha$, margin $m$, C-MICL $\alpha$
      ($1-$target), nominal none. \textbf{Robust~Reg.\ absent}: its dial \emph{is} $\rho$.
    \item Derived: \textbf{best objective} among delivering cells; assumes nothing.
    \item $\tau$'s grid \textbf{fixed before the run}: half-decades
      $10\!\to\!10^{-4}$, same at every $\rho$. A \emph{parameter}, never read back
      off the run; the bottom is the \texttt{mip\_gap} floor, below which $\tau$
      would be \emph{mislabelled} rather than smaller.
    \item Every grid spans its dial's \textbf{full usable range}: wrapper $\alpha$
      to $0.95$ ($1$ of $P$ --- $1.0$ deletes the constraint), $m$ to $5$, C-MICL
      $\alpha$ to $1$, down to what $n_{\text{cal}}$ certifies --- $1/(n_{\text{cal}}{+}1)$,
      so $0.02$ on gastric, $0.0075$ on the reactor. Search makes length cheap; the
      old grids bound 3 series and left C-MICL no point at all.
    \item \textbf{Searched, not walked}: ordered by robustness --- feasibility $0$
      prunes everything less robust, the solved floor everything more. Objective and
      solvability are monotone \emph{structurally}: each dial nests the feasible set.
      \mc{Feasibility is not: replaying 11 committed series, 6 dips, none across
      the target, $\text{dial}^\ast$ unchanged 11/11.}
  \end{itemize}
\end{frame}

\begin{frame}{Calibration: the $\rho$ axis (supporting)}
  Sweep the \textbf{shared} $\rho$; each method's own parameter \emph{fixed}
  ($\tau{=}0.01$, $\alpha{=}0.2$).
  \begin{itemize}
    \item $\rho^\ast(\text{method})$: largest $\rho$ meeting the feasibility target
      --- \emph{capacity}, not a frontier point.
    \item Curve is the primary reading; $\rho^\ast$ is \emph{derived} from it,
      re-derivable at a new target without re-solving.
    \item \textbf{Nothing is evaluated at $\rho^\ast$.} It answers ``how much
      assumed uncertainty does this method absorb'' --- a question about
      $\mathcal D$. The decisions are tested at $\text{dial}^\ast$, on the primary
      axis, two slides on.
    \item $\tau,\alpha$: ablations at one $\rho$.
  \end{itemize}
  Both axes: train-only; \mc{one bank per $(\rho$ column, fold$)$ serves every
  dial grid;} folds temporal \mc{(gastric)} / KFold; scores
  \textbf{conditional} on a solve with \texttt{solved\_frac} beside them; capped
  cells flagged. \mc{Per-context records $(\text{fold},\text{context},\text{solved},
  \text{feasible},\text{objective})$ are written too, so a same-cohort objective is
  derivable --- a conditional mean flatters whoever solved least.} Clock split
  \emph{master} vs \emph{test-point}: CP pays up front, the wrapper per point.
\end{frame}

\begin{frame}{The test stage: the sweep tunes, this tests}
  \small
  Each method held at its own $\text{dial}^\ast$, re-scored under a judge that dial
  never faced.
  \begin{itemize}
    \item Judge: analytic $f_{\text{true}}$ (synthetic); the \textbf{ODE}
      (reactor); the \textbf{full-cohort ($416$-arm) GT ensemble} on the $96$
      held-out arms (gastric) --- \emph{not} the $320$-arm tuning proxy. Both
      judge and cohort change there.
    \item \textbf{folds}: the sweep's folds re-solved. A rate with spread, but
      \emph{not} held out --- $\text{dial}^\ast$ was chosen on them.
    \item \textbf{full}: one refit on all rows --- the deployed procedure. One
      \emph{bit} of feasibility and one $x^\ast$ each, so it is where the
      objectives compare with nothing conditioned away.
    \item \textbf{subsample} (gastric only): \textbf{full} over $R{=}10$
      $m$-out-of-$n$ draws ($m/n{=}0.5$ of the fit rows, CRN, fixed oracle).
      Mean, \textbf{worst case}, 10th pct --- for feasibility \emph{and} the
      objective, each also over the \textbf{samestore} cohort (the arms every
      series solved that draw), which is the pair that compares. Gastric needs it because its judge is
      \emph{fitted} and its test is a cohort split, so the training draw is the
      only axis neither fixed by construction nor tuned on; the reactor is judged
      by the \textbf{ODE}, so its folds are already out of sample.
  \end{itemize}
\end{frame}

\begin{frame}{The test stage: what it refuses to report}
  \small
  \begin{itemize}
    \item \textbf{No $\text{dial}^\ast$ $\Rightarrow$ the series is skipped},
      not quietly tested at some other dial: nothing was tuned, so there is
      nothing to test.
    \item \texttt{--fallback-best-feas} tests it anyway at its
      \emph{most-feasible} cell, marked $\texttt{kind=best\_feas}$. That is a
      \textbf{ceiling} --- ``how feasible does it get at its best, under a fresh
      judge'' --- and \emph{never} a protocol point: unlike $\text{dial}^\ast$
      (best objective \emph{among} cells that already cleared the target), this
      dial is the argmax of the very quantity being re-reported. Never read a
      $\texttt{best\_feas}$ row against a $\texttt{tested}$ one as if the two
      were tuned alike.
    \item A series whose every dial fell under \texttt{--min-solved} has no cell
      to fall back \emph{to}, so its \emph{tuned} row is skipped regardless ---
      gastric C-MICL, where the fix is a wider $\alpha$ grid. But C-MICL is
      \textbf{always} tested at the $\alpha$ it \emph{asserts}
      ($1-$target$=0.1$, \texttt{kind=protocol}), in addition to any tuned row
      and even where there is none: that level is \textbf{pinned}, not fitted,
      so it is the one C-MICL row that reads directly on its own claim.
    \item \textbf{A fitted judge cannot score a boundary optimum} --- the proxy
      agrees with the ODE on $5/5$ nominal decisions yet flips $25.9\%$ of
      verdicts in its own $|F-50|<5$ band. Now \emph{instrumented}: both stages
      record per-decision \texttt{slack}, leave-one-member-out verdict flips and
      $x^\ast$, and \texttt{audit\_judge.py} reports
      $[\text{feas}_{\text{lo}},\text{feas}_{\text{hi}}]$ at $\kappa$ unexplained
      sd. \textbf{Reporting only}: $\text{dial}^\ast$ stays on the point verdict.
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Fix the model class first.} CV picks type + hyperparameters, then
      freezes them --- so model choice cannot confound the comparison, and
      $\mathcal D$'s radius is estimated with a pre-committed model.
    \item \textbf{One shared $\mathcal D$ \emph{on the sweep}}: a ball of radius
      $\rho$ unexplained sd, sphere draws, coherence as a grouping. At a given
      $\rho$ the methods differ in \emph{where} robustness enters, not in the
      uncertainty each was handed.
    \item \textbf{One dial each.} Primary axis: $\rho$ fixed, every method walked
      along its own dial, and the deliverable is the \emph{curve} in objective
      $\times$ feasibility. Supporting axis: $\rho$ swept at fixed dials, giving
      $\rho^\ast$ --- a capacity, reported, never evaluated at.
    \item \textbf{Test at $\text{dial}^\ast$ against a judge the dial never
      faced}, and on gastric over repeated subsampled training draws, on the
      cohort every method could prescribe for.
  \end{enumerate}
\end{frame}

\begin{frame}{C-MICL --- calibrate instead of assuming $\mathcal D$}
  \small
  Ovalle et al.\ 2025. Same goal --- ground-truth feasibility --- from held-out
  residuals rather than an assumed set. \texttt{src/methods/cmicl.py}.
  \begin{itemize}
    \item Split the training rows. Fit $\hat h$ (the embedded model) on
      proper-train; fit $\hat u$ there on $|y-\hat h(x)|$; on the \emph{held-out}
      calibration rows take $\hat q=s_{(k)}$,
      $k=\lceil(n_{\mathrm{cal}}{+}1)(1-\alpha)\rceil$, of
      $s_i=|y_i-\hat h(x_i)|/\hat u(x_i)$ --- $\hat u$ \emph{raw}, no floor.
    \item Embed \emph{both}:
      $\sum_m[w_m\hat h_m(x)+|w_m|\hat q_m\hat u_m(x)]\le b$, intersected with
      $[\underline y,\overline y]$, and $\hat u_m(x)\ge0$.
    \item \textbf{$\alpha$ pinned, not tuned}: $\alpha=1-$target$=0.1$. Others
      \emph{search} for the $\rho$ delivering $0.9$; C-MICL \emph{asserts} it. So
      it enters at \textbf{evaluation}, not on the $\rho$ axis.
    \item \textbf{Their implementation} (\texttt{github.com/dovallev/c-micl},
      \texttt{b44fe53}), not a port: $80/20$; $\hat u$ a shared $2\times32$ ReLU
      MLP, \emph{linear} output, not CV'd. Diff in \texttt{cmicl.py}: $\hat q$
      one order statistic apart, $\hat h$ hand-fixed there vs CV'd here.
  \end{itemize}
  \smallskip
  \textbf{Stated, not fixed}: marginal guarantee while $x^\ast$ is an argmin
  \emph{on} the constraint; gastric folds \emph{temporal}. Linear output
  $\Rightarrow\hat u<0$ possible, so $\hat u\ge0$ binds and $x^\ast$ sits on it
  ($\hat u(x^\ast)=10^{-15}$): the tightening is walked off, leaving nominal's
  $\hat h\ge b$ \emph{plus} an artifact restriction to $\{\hat u\ge0\}$.
\end{frame}

\begin{frame}{Tuned nominal --- the baseline to beat}
  \small
  Nominal fit, nominal MIP, RHS moved in. \texttt{src/methods/margin.py}.
  \vspace{-0.4em}
  \[
    \sum_m w_m f_m(x)\ \le\ b - m_c,
    \qquad m_c = m\,\sum_m |w_m|\,\mathrm{scale}(y_m)
  \]
  \vspace{-1.3em}
  \begin{itemize}
    \item \textbf{One dial $m$ per problem}, dimensionless. Each constraint is
      scaled by its \emph{own} $\mathrm{scale}(y_c)$ --- the same out-of-fold
      residual sd that sets $R_c$ --- so $m$ reads on the $\rho$/$\tau$ axis:
      $m{=}1$ is one unexplained sd of headroom per constraint.
    \item Gastric $m{=}1$: $b{=}0.6\to 0.35$ (DLT), $0.31$ (constitutional).
      $|w_m|$ keeps the sign: reactor $-F\le-50$ tightens to $F\ge52.19$.
    \item $m{=}0$ \emph{is} nominal (verified bit-identical, both problems).
    \item \textbf{Monotone} $\Rightarrow$ $m^\ast$ always exists at any target.
      Contrast robust\_reg, whose gastric feasibility \emph{falls} with $\rho$.
    \item Objective term deliberately untouched: a margin there is a
      \emph{constant}, shifting the reported objective without moving $x^\ast$.
  \end{itemize}
  \smallskip
  \textbf{Why it is here}: it is the cheapest thing that buys feasibility. A
  shared-$\mathcal D$ curve that does not sit above it at equal feasibility is
  paying for machinery a one-line RHS shift delivers. \textbf{No guarantee of any
  kind} --- $m$ is fitted to held-out feasibility, exactly as a fitted $\rho$ is.
