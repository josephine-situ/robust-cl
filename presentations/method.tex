\begin{frame}{Formulation}
  \textbf{Nominal}:
  \[
    \min_{x\in\mathcal{X}}\; c^\top x \quad\text{s.t.}\quad \hat f(x;\theta^\ast)\le b .
  \]
  \begin{itemize}
    \item Noisy labels $\Rightarrow$ different $\theta^\ast(\delta)$ $\Rightarrow$ infeasible
      decisions.
    \item \textbf{Robust}: feasible against \emph{every} reachable model.
  \end{itemize}
  \[
    \min_{x\in\mathcal{X}} c^\top x
    \;\;\text{s.t.}\;\; \max_{\delta\in\mathcal{D}} f(x;\theta^\ast(\delta))\le b,
  \]
  \[
    \theta^\ast(\delta)=\arg\min_{\theta\in\Theta_{\text{feas}}}\mathcal L(\theta;X,y+\delta)
  \]
\end{frame}

% ===========================================================================
\begin{frame}{Summary}
  \begin{enumerate}
    \item Select model class (type + hyperparameters).
    \item For each method:
      \begin{enumerate}
        \item Calibrate one robustness parameter ($\rho$) per method on held-out
              folds. Ablate other method-specific parameters.
        \item Evaluate fixed parameters on repeated subsampled training draws,
              against a fixed ground truth.
      \end{enumerate}
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Trees enter the MIP exactly\implnote}
  \begin{itemize}
    \item Leaf boxes separated by a $\le10^{-5}$ band on \emph{both} sides of each
      split, clamped so no training row loses its leaf. Verified
      $\le3\times10^{-7}$ vs the trained model, over all rows and perturbed refits.
    \item Both sides needed: a bound left at $\theta$ is still met at
      $\theta-$\texttt{FeasibilityTol}, routing $x$ to the wrong leaf.
    \item Branches routed in \texttt{float32} --- both libraries traverse there,
      so $x$ within an ulp of $\theta$ goes the way the model sends it.
    \item \texttt{IntFeasTol}$=10^{-9}$: big-M turns integrality slack into
      $M\!\cdot\!$\texttt{IntFeasTol} of $x$-slack ($M\!\approx\!46$). At the
      $10^{-5}$ default a free-$x$ adversary extracts $1.6\times10^{-2}$.
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: synthetic}
  True constraint, known exactly (evaluation only):
  \[
    f_{\text{true}}(x) = \sum_{j=1}^d x_j^2 + 0.5\prod_{j=1}^d x_j, \qquad x\in[0,1]^d
  \]
  Noisy labels $y_i = f_{\text{true}}(x_i)+\varepsilon_i$,
  $\varepsilon_i\sim\mathcal N(0,\sigma^2)$:
  \[
    \min_{x\in[0,1]^d} \; -\sum_{j=1}^d x_j \quad\text{s.t.}\quad \hat f(x;\theta)\le b,
    \qquad b=0.5\,d
  \]
  \begin{itemize}
    \item One LP, one learned constraint, no context.
    \item $n=200$, $d=2$, $\sigma=0.1$. Coherence vacuous (one outcome).
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: gastric}
  \small
  \textbf{A row is a published trial arm.} Meta-analysis of gastric-cancer
  chemotherapy studies; $z$, $x$ and labels are all arm-level \emph{aggregates}.
  \begin{itemize}
    \item \textbf{416 arms} by publication year:
      \textbf{320 train} ($\le$ 2008) / \textbf{96 test} (2009--2012).
    \item $x$: drug doses, in the hull of observed arms; one LP per arm.
    \item $y$: \% of the arm with grade-3/4 toxicity, per category; median OS.
    \item All 6 models fit the \emph{same} 320 arms $\Rightarrow$ one relabeling
      $\delta$ can move them together (\textbf{coherent}); \textbf{incoherent}
      gives each its own $\delta_c$. OS always independent.
  \end{itemize}
  \[
    \max_{x\,\in\,\mathrm{hull}(X_\text{train})} \widehat{\text{OS}}(x,z;\theta_\text{os})
    \quad\text{s.t.}\quad \widehat{\text{pctl}}_c(x,z;\theta_c) \le 0.6
  \]
  \[
    c\in\{\text{DLT, Blood, Constitutional, Infection, GI}\}
  \]
  \textbf{$z$ --- 9 arm-level covariates}, fixed within an arm:
  \begin{center}\footnotesize
  \begin{tabular}{@{}lll@{}}
    \texttt{Pub\_Year} & \texttt{FRAC\_MALE} & \texttt{Primary\_Stomach} \\
    \texttt{Asia} & \texttt{AGE\_MED} & \texttt{Primary\_GEJ} \\
    \texttt{N\_Patient} & \texttt{Prior\_Palliative\_Chemo} & \texttt{ECOG\_MEAN} \\
  \end{tabular}
  \end{center}
  \footnotesize
  All \emph{arm fractions} except \texttt{Pub\_Year}, \texttt{N\_Patient}
  (11--521), \texttt{AGE\_MED}, \texttt{ECOG\_MEAN} (0--4).
\end{frame}

\begin{frame}{Model class is chosen \emph{once}, before any robustness}
  \small
  5-fold CV on the \textbf{training} split; grid over
  \{linear (ElasticNet), SVM, CART, RF, GBM, XGB, MLP\}; score $=R^2$.
  \begin{itemize}
    \item Winner (type \emph{and} hyperparameters) frozen per learned model
      (5 constraints + OS).
    \item Gastric: XGB for DLT / Blood / Constitutional / Infection; ElasticNet for GI and OS.
    \item Ensures the model is \emph{fixed} across all robustness methods and does not confound.
    \item Also affects $\mathcal D$: its radius is estimated with this frozen model.
  \end{itemize}
\end{frame}

\begin{frame}{Robust methods summary}
  \small
  {\footnotesize\mc{Red} $=$ beyond the basic setting: needed only for multiple
  constraints / contexts / a learned objective (gastric).}
  \[
    \mathcal D_c = \{\delta:\ \lVert\delta\rVert_2\le R_c\},
    \qquad R_c = \rho\,\mathrm{scale}(y_c)\,\sqrt n
  \]
  \begin{itemize}
    \item \mc{Each $y$ has its own $\mathcal D_c$ (radius $R_c$), but the \emph{same} $\rho$.}
    \item While each method has its own tunable parameters, we only tune $\rho$ for
      consistency between methods.
    \item CP/wrapper embed scenarios as multiple \emph{constraints}.
      Embedding multiple directed scenarios makes the MIP infeasible.
  \end{itemize}
  \begin{center}\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    Method & Worst case found by & Scenario becomes \\
    \midrule
    Robust Reg. & \textbf{directed}: $\delta = R\,r/\lVert r\rVert$ & the \emph{fit} \\
    CP          & \textbf{random} draws, cut the argmax  & a \emph{cut} \\
    Wrapper     & \textbf{random} draws, chance constraint & $P$ \emph{constraints} \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{$\rho$: one size parameter, swept not tuned}
  \small
  \begin{itemize}
    \item \textbf{One parameter}, not two like a budget uncertainty set.
    \item $\sqrt n$: $\rho=1$ = L2 norm of iid noise at one
      unexplained sd/row. Transfers across $n{=}200$ and $n{=}320$.
    \item \textbf{Derived}: $\rho^\ast(\text{method})$ at a fixed feasibility target.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Radius = unexplained spread}
  \begin{center}\small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & $\mathrm{sd}(y)$ & OOF residual sd & unexplained \\
    \midrule
    Gastric toxicities & 0.288 & 0.249--0.293 & 0.78--1.02 \\
    Synthetic ($\sigma_{\text{true}}=0.100$) & 0.545 & \textbf{0.128} & 0.235 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item $\mathrm{sd}(y)$ is mostly \emph{signal}: on synthetic it is $4\times$ the
      unexplained spread, so the same $\rho$ inflates $\mathcal D$ $4\times$ ---
      CP would not converge.
    \item \texttt{scale\_stat=oof\_sd}: $\rho=1$ = \textbf{one unexplained sd per row}.
      Synthetic $0.128$ vs true noise $0.100$, without being told the DGP.
    \item Folds are the problem's own \mc{(temporal on gastric $\Rightarrow$ no
      leakage)}; a CV fold / subsample re-derives cutoffs and scale from
      \emph{its own} rows.
  \end{itemize}
\end{frame}

\begin{frame}{The scenario bank --- $\mathcal D$, made finite\implnote}
  \small
  $B$ draws from $\mathcal D$; one model retrained per draw \mc{per outcome}.
  \begin{itemize}
    \item Draw $b$ is a \textbf{pure function of $(\texttt{seed},b)$}
      $\Rightarrow$ a bank of size $P<B$ is a genuine \emph{prefix} of one of size $B$.
    \item \texttt{random\_state} fixed across members: the scenario is the
      \emph{only} source of variation.
    \item \mc{Coherent: one standardized direction shared across the \emph{group},
      scaled by each outcome's own radius. Incoherent: independent draws.}
    \item \mc{Percentile labels are clipped to $[0,1]$ (clipping only shrinks
      $|\delta_i|$, so the draw stays in $\mathcal D$).}
    \item \textbf{Built once}, not per iteration: with the draws fixed,
      $\theta(\delta_b)$ no longer depends on $x^\ast$, so CP's
      worst-violation-over-the-bank is monotone and $\tau$ means something.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\mc{Coherent is a grouping (gastric)}}
  \small
  Coherent worst-case draws asserts residual correlation $+1$ across outcomes;
  incoherent asserts $0$. OOF residuals ($n=145$, forward-chaining):
  \begin{center}\small
  \begin{tabular}{@{}lcc@{}}
    \toprule
    & percentile labels & raw labels \\
    \midrule
    Toxicity pairs (excl.\ DLT) & $+0.28$ & $+0.22$ \\
    OS vs.\ every toxicity      & $+0.06$ & $+0.02$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item \textbf{OS is uncoupled}.
    \item $\Rightarrow$ \texttt{coherent\_exclude}: toxicities share a direction,
      OS draws independently. A \emph{grouping}, not a flag.
    \item DLT excluded \emph{from the average}: $\text{DLT}=1-\prod_c(1-y_c)$
      holds exactly, so its $+0.44$--$0.80$ row is construction.
    \item But DLT stays \emph{in} the coherent group --- only OS is excluded.
      So: 5 outcomes' radius on 4 d.o.f., and no $\delta\in\mathcal D$ preserves
      the identity (percentile labels shift independently).
    \item Still \emph{asserted}: truth is $+0.28$, we impose $+1$ in-group.
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression --- robustify the \emph{fit}}
  \[
    \min_\theta \max_{\delta\in\mathcal D_c} \mathcal L(\theta; X, y_c+\delta)
  \]
  \begin{itemize}
    \item \textbf{Linear and one constraint}: exact SOCP, solved once (next slide).
    \item \textbf{Trees / ensembles}: no closed form --- alternate worst $\delta$ /
      retrain, $K=5$ times.
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression: the linear counterpart is an exact SOCP\implnote}
  \small
  On the ball the inner max is closed form --- push every residual outward along
  itself, $\delta^\ast = R\,r/\lVert r\rVert$:
  \[
    \max_{\lVert\delta\rVert_2\le R}\lVert r+\delta\rVert_2^2
      = \big(\lVert r\rVert_2 + R\big)^2,
    \qquad r = y - Z\beta - b_0 .
  \]
  Epigraph $\nu \ge \lVert r\rVert_2$ (one second-order cone) gives
  \[
  \begin{aligned}
    \min_{\beta,b_0,r,\nu}\;\; & \tfrac{1}{2n}\big[\nu^2 + 2R\,\nu + R^2\big]
      + \alpha\lambda\lVert\beta\rVert_1
      + \tfrac{\alpha}{2}(1-\lambda)\lVert\beta\rVert_2^2 \\
    \text{s.t.}\;\; & r_i = y_i - \beta^\top z_i - b_0, \quad i=1..n, \qquad
                      \textstyle\sum_i r_i^2 \le \nu^2 .
  \end{aligned}
  \]
  \begin{itemize}
    \item Relaxation is \textbf{tight}: $(\nu+R)^2$ increases in $\nu\ge0$ and we
      minimize, so the optimum never buys slack it is charged for.
    \item \textbf{Simpler than the box$\cap$L1 case}, which needed the top-$m$ sum
      $S_m(|r|)$ and its $2n{+}1$ LP epigraph variables. Here: one cone.
    \item Convex QP $\to$ convex SOCP. Penalty still matches sklearn's ElasticNet
      scaling $\Rightarrow$ comparable to the nominal fit; $R=0$ recovers it.
  \end{itemize}
\end{frame}

\begin{frame}{Wrapper --- chance-constrain a sample of $\mathcal D$}
  \begin{itemize}
    \item $P=20$ draws from the \emph{shared bank} (the sphere $\lVert\delta\rVert_2=R_c$, not
      bootstrap resamples); $P$ replicates of each learned model, \textbf{all embedded}.
    \begin{itemize}
      \item Differs from bootstrap resamples (prop 0.5) used by Maragno et al. for WFP example as
        it is smoother at small n. Also aligns with other robust methods.
    \end{itemize}
    \item \mc{Coherent: $z_p=1$ iff $x$ satisfies \emph{every} constraint under replicate
      $p$ (one indicator per draw). Incoherent: $z_{c,p}$ per constraint.}
    \item Chance constraint: $\tfrac1P\sum_p z_p \ge 1-\alpha$.
    \item \mc{OS robustification \textbf{off} by default --- matches CP.}
    \item Parameter: $\alpha$. $P$ is capped by MIP size.
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes (ours)}
  \small
  \begin{enumerate}
    \item Solve master $\to x^\ast$ \mc{--- gastric: one \emph{shared} master, solved
      per \textbf{anchor} (10 k-medoids over \emph{training} contexts - each test point
      is solved using the same master)}.
    \item \textbf{Separate}: score bank draws at $x^\ast$; cut the worst if its
      distance $>$ tolerance. Repeat.
  \end{enumerate}
  \begin{itemize}
    \item \mc{\textbf{coherent} (gastric): a scenario is one shared relabeling training
      all 5 constraints \emph{and} the objective; rank by \textbf{normalized average
      distance} (mean relative exceedance over anchor $\times$ outcome), cut the
      worst \emph{scenario}.}
    \item \mc{\texttt{cut\_whole\_scenario}: cut \emph{all} of an accepted scenario's
      constraints, not the breached subset --- makes per-scenario exclusion sound,
      and matches the wrapper's per-replicate indicator.}
    \item \mc{\textbf{OS robustification off}: ablation showed worse test feasibility
      \emph{and} OS.}
    \item $\tau$ (stopping tolerance) in units of unexplained SD.
    \item Separation \emph{pool}: $B{=}200$ scored, one cut/iter; wrapper embeds all $P{=}20$.
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes: what makes it terminate\implnote}
  \small
  Each of these was found by a run, not by reading. Together: gastric went from an
  exact period-4 cycle to \texttt{status=optimal} in 19 iterations.
  \begin{enumerate}
    \item \textbf{Fixed bank} (\texttt{scenario\_source: noise}). Redrawing each
      iteration compared iteration-$k$'s sample against iteration-0's $d_0$.
    \item \textbf{MIP gap $10^{-4}$} in the cut loop (was 1\%). Gastric obj
      ${\approx}10$ $\Rightarrow$ gap ${\approx}0.1$ vs distances ${\approx}0.007$:
      cuts below the solver's tolerance left $x^\ast$ unmoved. Synthetic never hit
      this (obj ${\approx}1.2$, distances ${\approx}0.1$).
    \item \textbf{Nothing is removed} from the master: pruning off,
      \texttt{cut\_eviction: reject}. Removing a cut lets a previous $x^\ast$ recur
      --- that is what a cycle is. Eligible set strictly shrinks
      $\Rightarrow$ terminates in $\le B$.
    \item \mc{\textbf{Protected anchor set is fixed}, not recomputed: the anchors the
      \emph{nominal} fit could serve (8/10 on gastric), tested \textbf{set-wise}.}
    \item \mc{\textbf{Rejections cached} --- sound given (3)+(4). Without it, rollbacks
      grew $8\to44$ per iteration.}
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\mc{Cutting Planes: coverage (anchors)}\implnote}
  \begin{itemize}
    \item Adding cuts can make the master infeasible at some anchor.
    \item A cut breaking the protected set is \textbf{rolled back and its draw
      permanently rejected}. So $\tau$ is CP's only robustness parameter.
    \item The protected set is \textbf{fixed}, not recomputed per iteration ---
      the anchors the \emph{nominal} fit could serve (8/10 on gastric), tested
      set-wise so CP cannot trade one patient's feasibility for another's.
    \item Rejections are \textbf{cached} (113/200 draws on gastric); a fixed
      protected set plus a monotone master makes that sound.
  \end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{Calibration: the target is $\rho^\ast$}
  Held-out CV, after model selection. Sweep the \textbf{shared} $\rho$; each
  method's own parameter \emph{fixed} ($\tau{=}0.01$, $\alpha{=}0.2$).
  \begin{itemize}
    \item $\rho^\ast(\text{method})$: largest $\rho$ meeting the feasibility target.
    \item Curve is the primary reading; $\rho^\ast$ is \emph{derived} from it,
      re-derivable at a new target without re-solving.
    \item \textbf{Evaluation runs each method at its own $\rho^\ast$}: matched
      held-out feasibility, not a matched $\mathcal D$.
    \item $\tau,\alpha$: ablations at one $\rho$.
    \item Train-only. Folds: temporal \mc{(gastric)} / KFold. Per (method,
      \mc{coherence}) cell.
    \item Scores \textbf{conditional} on a solve; \texttt{solved\_frac} beside
      them; capped cells out of $\rho^\ast$.
    \item Clock split \emph{master} vs \emph{test-point}: CP pays up front, the
      wrapper per point.
  \end{itemize}
\end{frame}

  \begin{itemize}
    \item Each method at its own $\rho^\ast$ (from the sweep), fixed before the draws.
    \item Uncertainty over \emph{training draws} --- distinct from the inner
      $\mathcal D$ the methods use. GT ensemble is a fixed oracle.
    \item GT is fit on a \emph{superset} of the constraint rows $\Rightarrow$ a
      single full-data draw favours nominal. Hence subsampling.
    \item Common random numbers across methods; \texttt{all\_constraints} only.
  \end{itemize}
  \smallskip
  \footnotesize
  \textbf{Samestore cohort} (gastric): every number, feasibility and OS alike, is a
  mean over the test arms \textbf{every} method could prescribe
  for, recomputed per draw $\Rightarrow$ feasibility is \textbf{conditional on
  solvability}, reported separately.
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Fix the model class first.} CV picks type + hyperparameters, then
      freezes them --- so model choice cannot confound the comparison, and
      $\mathcal D$'s radius is estimated with a pre-committed model.
    \item \textbf{One shared $\mathcal D$ \emph{on the sweep}}: a ball of radius
      $\rho$ unexplained sd, sphere draws, coherence as a grouping. At a given
      $\rho$ the methods differ in \emph{where} robustness enters, not in the
      uncertainty each was handed.
    \item \textbf{One parameter each}, held fixed while $\rho$ is swept. Evaluation
      then runs each method at its own $\rho^\ast$: matched held-out feasibility,
      \emph{not} a matched $\mathcal D$.
    \item \textbf{Evaluate against a fixed ground truth} over repeated subsampled
      training draws, on the cohort every method could prescribe for.
  \end{enumerate}
\end{frame}

\begin{frame}{Related: conformal MICL (\textbf{not implemented})}
  \small
  Ovalle et al.\ 2025. Same goal --- ground-truth feasibility --- by conformal
  prediction rather than an ensemble.
  \begin{itemize}
    \item Fit $\hat h$; fit $\hat u$ on $|\hat h(x)-y|$; take $\hat q_{1-\alpha}$
      of $|\hat h-y|/\hat u$ on a \emph{held-out} calibration set. Embed
      $\hat h(x)\pm\hat q_{1-\alpha}\hat u(x)\subseteq[\underline y,\overline y]$.
    \item Guarantee is \textbf{marginal} over $\mathcal P_{XY}$ and rests on
      feasibility $\perp$ coverage $\mid$ GT feasibility. Our $x^\ast$ is an
      argmin, not a random draw.
  \end{itemize}
  \medskip
  \textbf{Why we expect it to be infeasible on gastric}: unexplained ratio
  $0.78$--$1.02$ $\Rightarrow$ $\hat u\approx\mathrm{sd}(y)$ $\Rightarrow$
  ${\approx}1.65\,\mathrm{sd}(y)$ half-width on 5 constraints at once. CP absorbs
  the same poor fit by cutting the worst scenario instead.
  \medskip
  \footnotesize
  Not implemented: no calibration split, no $\hat u$, and our folds are
  \emph{temporal} --- conformal needs exchangeability.
\end{frame}
