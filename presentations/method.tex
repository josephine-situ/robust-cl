\begin{frame}{Formulation}
  \textbf{Nominal}:
  \[
    \min_{x\in\mathcal{X}}\; c^\top x \quad\text{s.t.}\quad \hat f(x;\theta^\ast)\le b .
  \]
  \begin{itemize}
    \item Noisy labels $\Rightarrow$ different $\theta^\ast(\delta)$ $\Rightarrow$ infeasible
      decisions.
    \item \textbf{Robust}: feasible against \emph{every} reachable model.
  \end{itemize}
  \[
    \min_{x\in\mathcal{X}} c^\top x
    \;\;\text{s.t.}\;\; \max_{\delta\in\mathcal{D}} f(x;\theta^\ast(\delta))\le b,
  \]
  \[
    \theta^\ast(\delta)=\arg\min_{\theta\in\Theta_{\text{feas}}}\mathcal L(\theta;X,y+\delta)
  \]
  \begin{itemize}
    \item $\delta$ perturbs \emph{labels}; the model is \textbf{retrained}. Not a
      post-hoc perturbation of predictions.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Trees enter the MIP exactly}
  \begin{itemize}
    \item Leaf boxes separated by a $\le10^{-5}$ band on \emph{both} sides of each
      split, clamped so no training row loses its leaf. Verified
      $\le3\times10^{-7}$ vs the trained model, over all rows and perturbed refits.
    \item Both sides needed: a bound left at $\theta$ is still met at
      $\theta-$\texttt{FeasibilityTol}, routing $x$ to the wrong leaf.
    \item Branches routed in \texttt{float32} --- both libraries traverse there,
      so $x$ within an ulp of $\theta$ goes the way the model sends it.
    \item \texttt{IntFeasTol}$=10^{-9}$: big-M turns integrality slack into
      $M\!\cdot\!$\texttt{IntFeasTol} of $x$-slack ($M\!\approx\!46$). At the
      $10^{-5}$ default a free-$x$ adversary extracts $1.6\times10^{-2}$.
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: synthetic}
  True constraint, known exactly (evaluation only):
  \[
    f_{\text{true}}(x) = \sum_{j=1}^d x_j^2 + 0.5\prod_{j=1}^d x_j, \qquad x\in[0,1]^d
  \]
  Noisy labels $y_i = f_{\text{true}}(x_i)+\varepsilon_i$,
  $\varepsilon_i\sim\mathcal N(0,\sigma^2)$:
  \[
    \min_{x\in[0,1]^d} \; -\sum_{j=1}^d x_j \quad\text{s.t.}\quad \hat f(x;\theta)\le b,
    \qquad b=0.5\,d
  \]
  \begin{itemize}
    \item One LP, one learned constraint, no context.
    \item $n=200$, $d=2$, $\sigma=0.1$. Coherence vacuous (one outcome).
  \end{itemize}
\end{frame}

\begin{frame}{Problem settings: gastric}
  \small
  \textbf{A row is a published trial arm.} Meta-analysis of gastric-cancer
  chemotherapy studies; $z$, $x$ and labels are all arm-level \emph{aggregates}.
  \begin{itemize}
    \item \textbf{416 arms} by publication year:
      \textbf{320 train} ($\le$ 2008) / \textbf{96 test} (2009--2012).
    \item $x$: drug doses, in the hull of observed arms; one LP per arm.
    \item $y$: \% of the arm with grade-3/4 toxicity, per category; median OS.
    \item All 6 models fit the \emph{same} 320 arms $\Rightarrow$ one relabeling
      $\delta$ can move them together (\textbf{coherent}); \textbf{incoherent}
      gives each its own $\delta_c$. OS always independent.
  \end{itemize}
  \[
    \max_{x\,\in\,\mathrm{hull}(X_\text{train})} \widehat{\text{OS}}(x,z;\theta_\text{os})
    \quad\text{s.t.}\quad \widehat{\text{pctl}}_c(x,z;\theta_c) \le 0.6
  \]
  \[
    c\in\{\text{DLT, Blood, Constitutional, Infection, GI}\}
  \]
  \textbf{$z$ --- 9 arm-level covariates}, fixed within an arm:
  \begin{center}\footnotesize
  \begin{tabular}{@{}lll@{}}
    \texttt{Pub\_Year} & \texttt{FRAC\_MALE} & \texttt{Primary\_Stomach} \\
    \texttt{Asia} & \texttt{AGE\_MED} & \texttt{Primary\_GEJ} \\
    \texttt{N\_Patient} & \texttt{Prior\_Palliative\_Chemo} & \texttt{ECOG\_MEAN} \\
  \end{tabular}
  \end{center}
  \footnotesize
  All \emph{arm fractions} except \texttt{Pub\_Year}, \texttt{N\_Patient}
  (11--521), \texttt{AGE\_MED}, \texttt{ECOG\_MEAN} (0--4).
\end{frame}

\begin{frame}{Model class is chosen \emph{once}, before any robustness}
  \small
  5-fold CV on the \textbf{training} split; grid over
  \{linear (ElasticNet), SVM, CART, RF, GBM, XGB, MLP\}; score $=R^2$.
  \begin{itemize}
    \item Winner (type \emph{and} hyperparameters) frozen per learned model
      (5 constraints + OS).
    \item Gastric: XGB for DLT / Blood / Constitutional / Infection; ElasticNet for GI and OS.
    \item Ensures the model is \emph{fixed} across all robustness methods and does not confound.
    \item Also fixes $\mathcal D$: its radius is estimated with this frozen model.
  \end{itemize}
\end{frame}

\begin{frame}{One uncertainty set $\mathcal D$, three methods}
  \small
  Was: each method carried its own uncertainty notion, so method differences were
  partly input differences.
  \[
    \underbrace{\{\delta: |\delta_i|\le\varepsilon_c,\ \lVert\delta\rVert_1 \le
      \texttt{budget\_frac}\cdot n\,\varepsilon_c\}}_{\texttt{box\_l1}:\
      \varepsilon_c=\varepsilon_0\mathrm{scale}(y_c)}
    \qquad
    \underbrace{\{\delta: \lVert\delta\rVert_2\le R_c\}}_{\texttt{ellipsoid}:\
      R_c=\rho\,\mathrm{scale}(y_c)\sqrt n}
  \]
  \begin{center}\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    Method & Use of $\mathcal D$ & Dial \\
    \midrule
    Robust Reg. & robustifies the \emph{fit}       & \texttt{label\_eps} \\
    Wrapper     & chance-constrains a sample       & $\alpha$ \\
    CP (ours)   & separates \emph{lazily}          & $\tau$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item Size is a shared \textbf{constant}, not a knob. One dial per method.
    \item Draws are \textbf{vertices} (box) / uniform on the sphere (ellipsoid) ---
      where Robust Reg.'s inner max sits. No method gets a weaker adversary.
    \item \textbf{Coherence} is a \emph{grouping}, crossed with all three methods.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{$\rho$: one size parameter, swept not tuned}
  \small
  \begin{itemize}
    \item \texttt{budget\_frac} cannot constrain a ball --- no L1 face, no
      support restriction --- so it only scaled the radius:
      $(\varepsilon_0,\texttt{budget\_frac})$ was \textbf{non-identifiable}.
      $\rho$ replaces both, losing nothing.
    \item $\sqrt n$, not $\sqrt m$: $\rho=1$ = L2 norm of iid noise at one
      unexplained sd/row. Transfers across $n{=}200$ and $n{=}320$.
    \item \textbf{Not the old operating point}: $\sqrt2\times$ wider than
      $\varepsilon_0{=}1$, ball $2.0\times$ stronger $\Rightarrow\approx2.8\times$
      the old adversary. Expect $\rho^\ast\ll1$.
    \item \textbf{Swept, never selected.} Per-method $\rho^\ast$ dissolves
      shared-$\mathcal D$; global selection has no honest criterion (GT ensemble
      $=$ the judge; synthetic $\sigma_{\text{true}}$ $=$ the DGP).
    \item \textbf{Primary} reading: the whole axis, dials fixed.
      \textbf{Derived}: $\rho^\ast(\text{method})$ at a fixed feasibility target.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Radius = unexplained spread, not marginal}
  \small
  $\delta$ shifts labels \emph{before} retraining $\Rightarrow$ the radius lives in
  label space. Both candidates do; they measure different things.
  \begin{center}\small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & $\mathrm{sd}(y)$ & OOF residual sd & unexplained \\
    \midrule
    Gastric toxicities & 0.288 & 0.249--0.293 & 0.78--1.02 \\
    Synthetic ($\sigma_{\text{true}}=0.100$) & 0.545 & \textbf{0.128} & 0.235 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item $\mathrm{sd}(y)$ is mostly \emph{signal}. At $\varepsilon_0=1$ it corrupts
      synthetic $4\times$ too hard --- CP would not converge. Kept as an ablation.
    \item \texttt{scale\_stat=oof\_sd}: $\varepsilon_0=1$ = \textbf{one unexplained sd}.
      Synthetic $0.128$ vs true noise $0.100$, without being told the DGP.
    \item Folds are the problem's own (temporal on gastric $\Rightarrow$ no leakage).
    \item Caveat: conflates noise with misspecification. Gastric fits are poor, so
      $\mathcal D$ stays wide there. \textbf{No coverage claim.}
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{The scenario bank --- $\mathcal D$, made finite}
  \small
  $B$ draws from $\mathcal D$; one model retrained per draw per outcome.
  \begin{itemize}
    \item Draw $b$ is a \textbf{pure function of $(\texttt{seed},b)$}
      $\Rightarrow$ a bank of size $P<B$ is a genuine \emph{prefix} of one of size $B$.
    \item \texttt{random\_state} fixed across members: the scenario is the
      \emph{only} source of variation.
    \item Coherent: one standardized direction shared across the \emph{group},
      scaled by each outcome's own radius. Incoherent: independent draws.
    \item Percentile labels are clipped to $[0,1]$ (clipping only shrinks
      $|\delta_i|$, so the draw stays in $\mathcal D$).
    \item \textbf{Built once}, not per iteration: with the draws fixed,
      $\theta(\delta_b)$ no longer depends on $x^\ast$, so CP's
      worst-violation-over-the-bank is monotone and $\tau$ means something.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Coherence is a grouping --- measured, not asserted}
  \small
  Coherent asserts residual correlation $+1$ across outcomes; incoherent asserts
  $0$. Neither was ever checked. OOF residuals, same rows / folds / frozen
  configs the bank uses ($n=145$, forward-chaining):
  \begin{center}\small
  \begin{tabular}{@{}lcc@{}}
    \toprule
    & percentile labels & raw labels \\
    \midrule
    Toxicity pairs (excl.\ DLT) & $+0.28$ & $+0.22$ \\
    OS vs.\ every toxicity      & $+0.06$ & $+0.02$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item \textbf{OS is uncoupled} --- 3 of 5 pairs negative on raw. Coherent's
      $+1$ was the least defensible claim in $\mathcal D$.
    \item $\Rightarrow$ \texttt{coherent\_exclude}: toxicities share a direction,
      OS draws independently. A \emph{grouping}, not a flag.
    \item DLT excluded: $\text{DLT}=1-\prod_c(1-y_c)$, so its $+0.44$--$0.80$ row
      is construction, not evidence.
    \item Matches the story that licenses coherence --- under-reporting adverse
      events spans toxicity endpoints, never survival.
    \item Still \emph{asserted}: truth is $+0.28$, we impose $+1$ in-group.
  \end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{Robust methods: overview}
  Shared MIP scaffolding --- they differ in \emph{where} robustness enters.
  \begin{itemize}
    \item \textbf{Nominal} --- none. No knob.
    \item \textbf{Robust Regression} --- one label-robust model per outcome.
    \item \textbf{Wrapper} --- chance constraint over $P$ bank models.
    \item \textbf{Cutting Planes (ours)} --- cut worst-case models at $x^\ast$.
    \item One monotone knob each, swept for a Pareto frontier.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Robust Regression --- robustify the \emph{fit}}
  \[
    \min_\theta \max_{\delta\in\mathcal D_c} \mathcal L(\theta; X, y_c+\delta)
  \]
  \begin{itemize}
    \item \textbf{Knob}: $\varepsilon = \texttt{label\_eps}\cdot\mathrm{scale}(y_c)$ ---
      unitless, so one grid covers toxicity fractions and OS months.
    \item \textbf{Linear}: exact QP, solved once (next slide).
    \item \textbf{Trees / ensembles}: no closed form --- alternate worst $\delta$ /
      retrain, $K=5$ times.
    \item \textbf{Coherent} arm implemented: the adversary picks \emph{one} shared
      row set across outcomes. Incoherent solves per outcome.
    \item Same $\mathcal D$ family as CP / wrapper, \emph{its own radius}:
      $\texttt{label\_eps}^\ast = 0.75$ (gastric coh.) vs $\varepsilon_0=1$, so RR
      fits against a narrower set than the bank is sampled from.
  \end{itemize}
\end{frame}

\begin{frame}{Robust Regression: the linear counterpart is an exact QP}
  \small
  Inner max over $\mathcal D$ is attained at a vertex --- shift the $m$ largest
  residuals by $\varepsilon$ in their own sign:
  \[
    \max_{\delta\in\mathcal D}\lVert r+\delta\rVert^2
      = \lVert r\rVert^2 + 2\varepsilon\, S_m(|r|) + m\varepsilon^2,
    \qquad r = y - Z\beta - b_0,
  \]
  with $S_m$ = sum of the $m$ largest $|r_i|$ (convex, LP-representable; the
  $m\varepsilon^2$ is constant in $\theta$ and dropped). So
  \[
  \begin{aligned}
    \min_{\beta,b_0,r,a,q,t}\;\; & \tfrac{1}{2n}\Big[\textstyle\sum_i r_i^2
      + 2\varepsilon\big(m\,t + \sum_i q_i\big)\Big]
      + \alpha\lambda\lVert\beta\rVert_1
      + \tfrac{\alpha}{2}(1-\lambda)\lVert\beta\rVert_2^2 \\
    \text{s.t.}\;\; & r_i = y_i - \beta^\top z_i - b_0, \quad i=1..n \\
                    & a_i \ge r_i,\quad a_i \ge -r_i, \quad
                      q_i \ge a_i - t, \quad q_i \ge 0
  \end{aligned}
  \]
  \begin{itemize}
    \item $m\,t+\sum_i q_i$ is the LP epigraph of $S_m(|r|)$ at the optimum.
    \item Penalty matches sklearn's ElasticNet scaling $\Rightarrow$ directly comparable
      to the nominal fit; $\varepsilon=0$ recovers it.
  \end{itemize}
\end{frame}

\begin{frame}{Wrapper --- chance-constrain a sample of $\mathcal D$}
  \begin{itemize}
    \item $P=20$ bank draws (each from half of the full set); $P$ replicates of each learned model, \textbf{all embedded}.
    \item Coherent: $z_p=1$ iff $x$ satisfies \emph{every} constraint under replicate
      $p$ (one indicator per draw). Incoherent: $z_{c,p}$ per constraint.
    \item Chance constraint: $\tfrac1P\sum_p z_p \ge 1-\alpha$.
    \item OS robustification \textbf{off} by default --- matches CP, and the legacy
      epigraph-over-$P$ handed the wrapper conservatism CP does not carry.
    \item Knob: $\alpha$. $P$ is capped by MIP size, not by statistics.
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes (ours)}
  \small
  \begin{enumerate}
    \item Solve master $\to x^\ast$ --- gastric: one \emph{shared} master, solved per
      \textbf{anchor} (10 k-medoids over \emph{training} contexts).
    \item \textbf{Separate}: score bank draws at $x^\ast$; cut the worst if its
      distance $>$ tolerance. Repeat.
  \end{enumerate}
  \begin{itemize}
    \item \textbf{basic} (synthetic): score every not-yet-embedded draw at $x^\ast$
      with the \emph{actual} constraint model, cut the single worst.
    \item \textbf{coherent} (gastric): a scenario is one shared relabeling training
      all 5 constraints \emph{and} the objective; rank by \textbf{normalized average
      distance} (mean relative exceedance over anchor $\times$ outcome), cut the
      worst \emph{scenario}.
    \item \texttt{cut\_whole\_scenario}: cut \emph{all} of an accepted scenario's
      constraints, not the breached subset --- makes per-scenario exclusion sound,
      and matches the wrapper's per-replicate indicator.
    \item A cut is a full embedded model, valid at \emph{any} context $\Rightarrow$
      computed \textbf{offline}; test points enter only at prescribe time.
    \item \textbf{OS robustification off}: ablation showed worse test feasibility
      \emph{and} OS.
  \end{itemize}
\end{frame}

\begin{frame}{Cutting Planes: what makes it terminate}
  \small
  Each of these was found by a run, not by reading. Together: gastric went from an
  exact period-4 cycle to \texttt{status=optimal} in 19 iterations.
  \begin{enumerate}
    \item \textbf{Fixed bank} (\texttt{scenario\_source: noise}). Redrawing each
      iteration compared iteration-$k$'s sample against iteration-0's $d_0$.
    \item \textbf{MIP gap $10^{-4}$} in the cut loop (was 1\%). Gastric obj
      ${\approx}10$ $\Rightarrow$ gap ${\approx}0.1$ vs distances ${\approx}0.007$:
      cuts below the solver's tolerance left $x^\ast$ unmoved. Synthetic never hit
      this (obj ${\approx}1.2$, distances ${\approx}0.1$).
    \item \textbf{Nothing is removed} from the master: pruning off,
      \texttt{cut\_eviction: reject}. Removing a cut lets a previous $x^\ast$ recur
      --- that is what a cycle is. Eligible set strictly shrinks
      $\Rightarrow$ terminates in $\le B$.
    \item \textbf{Protected anchor set is fixed}, not recomputed: the anchors the
      \emph{nominal} fit could serve (8/10 on gastric), tested \textbf{set-wise}.
    \item \textbf{Rejections cached} --- sound given (3)+(4). Without it, rollbacks
      grew $8\to44$ per iteration.
  \end{enumerate}
\end{frame}

\begin{frame}{Cutting Planes: knobs}
  \small
  \textbf{$\tau$ --- stopping tolerance} (the robustness dial)
  \begin{itemize}
    \item In \textbf{unexplained sd}: exceedance $\div\,\mathrm{scale}(y_c)$ ---
      \emph{per outcome}, the same scale that sets $\mathcal D$'s radius ---
      meaned over (anchor $\times$ outcome) cells. Shares units with $\rho$.
    \item \textbf{Grid spans nominal}: $\tau$ above the iter-0 distance cuts
      nothing. Synthetic (scale $0.1281$, max $0.1259$): $\tau{=}1$ is nominal
      exactly. Legacy \texttt{d0}: no $\tau\le1$ was.
    \item Mean is \textbf{anchor-count stable} (numerator scales with
      denominator): gastric $n_a{=}4/8/16 \to 0.031/0.032/0.018$, no $1/n$.
      Breadth is the point --- 2 of 20 cells broken beats 20 of 20.
    \item One grid, \textbf{log-spaced}: coherent scores (violating fraction)
      $\times$ (mean exceedance), basic has 1 cell and no fraction --- gastric
      $0.03$ vs synthetic $0.98$. Range, not meaning.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Cutting Planes: coverage}
  \begin{itemize}
    \item Adding cuts can make the master infeasible at some anchor.
    \item A cut breaking the protected set is \textbf{rolled back and its draw
      permanently rejected}. So $\tau$ is CP's only robustness knob.
    \item The protected set is \textbf{fixed}, not recomputed per iteration ---
      the anchors the \emph{nominal} fit could serve (8/10 on gastric), tested
      set-wise so CP cannot trade one patient's feasibility for another's.
    \item Rejections are \textbf{cached} (113/200 draws on gastric); a fixed
      protected set plus a monotone master makes that sound.
  \end{itemize}
\end{frame}

\begin{frame}{Two structural results}
  \small
  \textbf{CP $\equiv$ lazy wrapper at $\alpha=0$}
  \begin{itemize}
    \item The wrapper's $P$ models are a nested \emph{prefix} of CP's bank.
    \item At $B=P$: CP ($\tau\to0$) and wrapper ($\alpha=0$) return the same
      $x^\ast$, $\max|\Delta x| = 0$.
  \end{itemize}
  \medskip
  \textbf{$10\times$ the adversaries, $1/5$ the MIP} (synthetic)
  \begin{center}\small
  \begin{tabular}{@{}lrrr@{}}
    \toprule
    & adversaries & vars & constrs \\
    \midrule
    Wrapper & $P=20$  & 31{,}917 & 101{,}172 \\
    CP      & $B=200$ & \textbf{6{,}380} & \textbf{20{,}236} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item Wrapper embeds all $P$; CP embeds one cut per iteration. The bank is a
      separation \emph{pool}, not an embedded ensemble.
  \end{itemize}
\end{frame}

\begin{frame}{Calibration: the target is $\rho^\ast$, not each knob}
  \small
  Held-out CV, run \emph{after} model selection. Under the $\rho$ parameterization
  it sweeps the \textbf{shared} $\rho$ with every method's own dial \emph{fixed}
  ($\tau{=}0.01$, $\alpha{=}0.2$):
  \begin{itemize}
    \item $\rho^\ast(\text{method})$ = largest $\rho$ still meeting the held-out
      feasibility target. How much assumed uncertainty each method absorbs.
    \item $\tau$, $\alpha$ become \textbf{ablations at one $\rho$} --- enough to
      show the fixed values were not cherry-picked.
    \item Scores \textbf{conditional} on the solve returning a prescription;
      \texttt{solved\_frac} beside them. \texttt{max\_iterations} cells excluded
      from $\rho^\ast$.
    \item \textbf{Train-only} oracle. Gastric: temporal forward-chaining folds;
      synthetic: KFold. Per (method, coherence) cell.
    \item $\rho$ is never \emph{selected} --- the curve is the result.
    \item Every cell records \textbf{status} + wall clock split: \emph{master}
      phase (for CP the whole cut loop) vs \emph{test-point} phase. CP pays up
      front then prescribes from a small master; the wrapper pays per point.
  \end{itemize}
  \footnotesize\emph{Legacy per-method $\kappa^\ast$ (d\textsubscript{0}-basis
  $\tau$): gastric coh.\ $0.10/0.75/0.30$, incoh.\ $0.10/0.25/0.20$. Superseded.}
\end{frame}

\begin{frame}{Evaluation}
  \small
  \begin{center}
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.4cm}>{\raggedright\arraybackslash}p{3.8cm}>{\raggedright\arraybackslash}p{3.6cm}@{}}
    \toprule
    & Gastric & Synthetic \\
    \midrule
    Ground truth & GT ensemble, fit on all \textbf{416} arms (train + test) & Exact analytic constraint \\[0.3em]
    New train draw & $m$ of $n=320$ arms, no replacement; headline $m/n=0.5$,
      $R=10$ draws & Fresh noise, seeded per draw \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{itemize}
    \item Uncertainty over \emph{training draws} --- distinct from the inner
      $\mathcal D$ the methods use. GT ensemble is a fixed oracle.
    \item \textbf{A single draw on all training data favours nominal}: GT is fit on
      a \emph{superset} of the constraint models' rows, so we subsample instead.
    \item Sweep axes at common random numbers: rhs $0.4$--$0.8$, scarcity
      $m/n\in[0.3,0.8]$. \texttt{all\_constraints} only.
  \end{itemize}
  \medskip
  \footnotesize
  \textbf{Samestore cohort} (gastric) --- every number, \emph{feasibility and OS
  alike}, is a mean over the test arms \textbf{every} method (and strength cell)
  could prescribe for, recomputed \emph{per draw}. Feasibility is therefore
  \textbf{conditional on solvability}, which is reported separately.
\end{frame}

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Fix the model class first.} CV picks type + hyperparameters, then
      freezes them --- so model choice cannot confound the comparison, and
      $\mathcal D$'s radius is estimated with a pre-committed model.
    \item \textbf{One shared $\mathcal D$}, at one unexplained sd, vertex draws,
      coherence as a flag. Methods differ in \emph{where} robustness enters, not in
      the uncertainty each was handed.
    \item \textbf{Calibrate one knob each} on held-out folds, per coherence cell
      $\Rightarrow$ a matched robustness level.
    \item \textbf{Evaluate against a fixed ground truth} over repeated subsampled
      training draws, on the cohort every method could prescribe for.
  \end{enumerate}
  \medskip
  \textbf{Structural}: CP is a lazy wrapper --- identical at $\alpha=0$, but
  $10\times$ the adversaries in $1/5$ the MIP.
\end{frame}

\begin{frame}{Related: conformal MICL (\textbf{not implemented})}
  \small
  Ovalle et al.\ 2025. Same goal --- ground-truth feasibility --- by conformal
  prediction rather than an ensemble.
  \begin{itemize}
    \item Fit $\hat h$; fit $\hat u$ on $|\hat h(x)-y|$; take $\hat q_{1-\alpha}$
      of $|\hat h-y|/\hat u$ on a \emph{held-out} calibration set. Embed
      $\hat h(x)\pm\hat q_{1-\alpha}\hat u(x)\subseteq[\underline y,\overline y]$.
    \item \textbf{Two models, not $P$}; formulation size independent of $N$.
    \item Guarantee is \textbf{marginal} over $\mathcal P_{XY}$ and rests on
      feasibility $\perp$ coverage $\mid$ GT feasibility. Our $x^\ast$ is an
      argmin, not a random draw.
    \item $\hat u$ is scale-free ($\hat q$ absorbs any uniform bias); only
      \emph{shape} error hurts.
  \end{itemize}
  \medskip
  \textbf{Why we expect it to be infeasible on gastric}: unexplained ratio
  $0.78$--$1.02$ $\Rightarrow$ $\hat u\approx\mathrm{sd}(y)$ $\Rightarrow$
  ${\approx}1.65\,\mathrm{sd}(y)$ half-width on 5 constraints at once. CP absorbs
  the same poor fit by cutting the worst scenario instead.
  \medskip
  \footnotesize
  Not implemented: no calibration split, no $\hat u$, and our folds are
  \emph{temporal} --- conformal needs exchangeability.
\end{frame}
