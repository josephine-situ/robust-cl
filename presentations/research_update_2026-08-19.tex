\begin{frame}{The $\rho$ sweep --- protocol}
  \small
  One shared $D$ per outcome $c$; every method faces it at each $\rho$.
  \vspace{-0.4em}
  \[
    D_c=\{\delta\in\mathbb{R}^{n}:\|\delta\|_2\le R_c\},\quad
    R_c=\rho\,\mathrm{scale}(y_c)\sqrt{n}
  \]
  \vspace{-1.2em}
  \begin{itemize}
    \item $\mathrm{scale}(y_c)$ = out-of-fold residual sd. $\delta$ added to
      $y_c$ \emph{before} training; model refit.
    \item Grid $\rho\in\{0.05,0.1,0.2,0.3,0.5,0.75,1\}$. $\rho{=}1$ = iid noise
      at one unexplained sd per row.
    \item Own dials \emph{fixed}: CP $\tau{=}0.01$, wrapper $\alpha{=}0.2$.
      Ablated separately (gastric $\rho{=}0.75$, synthetic $\rho{=}1$).
    \item Scored on CV folds of \emph{train} (4 temporal / 4 KFold). The real
      test set is not used here.
    \item \textbf{$\rho^*$ rule (derived, not fitted):} largest grid $\rho$ with
      held-out feasibility $\ge0.9$ \emph{and} solved fraction $\ge0.5$.
  \end{itemize}
\end{frame}

\begin{frame}{Feasibility Discussion}
  \small
  \begin{itemize}
    \item Robust reg becomes less robust as we increase $\rho$.
      Intuitively, this is because $\rho$ increases the weight on the residuals,
      rather than the penalty terms (for elastic net).
  \end{itemize}
  \vspace{-0.6em}
  \[
    \bar g(v)=\max_{\|\delta\|_2\le\rho}\|v+\delta\|_2^2=(\|v\|_2+\rho)^2,
    \qquad \delta^\star=\rho\,v/\|v\|_2
  \]
  \vspace{-1.4em}
  \[
    \min_f\;\bar g\big(y-f(X)\big)
    =\min_f\Big[\underbrace{\|y-f(X)\|_2^2}_{\text{ordinary MSE}}
      +\underbrace{2\rho\|y-f(X)\|_2}_{\text{new, }f\text{-dependent}}\Big]+\rho^2
  \]
  \vspace{-0.8em}
  \begin{itemize}
    \item Robust classification doesn't have the same problem because they look 
      at label flips, rather than regression.
    \item Gastric is very noisy. May need to run more draws.
  \end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{Feasibility Discussion --- two ways out}
  \small
  \textbf{A. Drop robust\_reg.} The tuned-nominal baselines cover the slot:
  RHS margin $f(x)\le b-m$, or a penalty multiplier $c\,\alpha$.
  \vspace{-0.2em}
  \begin{itemize}\footnotesize
    \item Margin monotone in $m$, so $m^\star$ always exists. $c\,\alpha$ is
      robust\_reg reversed: $\alpha_{\mathrm{eff}}=\alpha/(1+\rho/\|r^\star\|_2)$
      only ever \emph{shrinks} $\alpha$.
  \end{itemize}
  \medskip
  \textbf{B. Change the shared $D$ to budgeted} --- still one set, one dial, all methods:
  \vspace{-0.5em}
  \[
    D_c=\{\delta:\ |\delta_i|\le\varepsilon_c,\ \|\delta\|_1\le m\varepsilon_c\},
    \quad \varepsilon_c=\rho\,\mathrm{scale}(y_c),\ m=\phi n
  \]
  \vspace{-1.5em}
  \[
    \bar g(v)=\|v\|_2^2
      +2\varepsilon\underbrace{S_m(|v|)}_{\text{sum of }m\text{ largest }|v_i|}
      +m\varepsilon^2
  \]
  \vspace{-0.6em}
  \begin{itemize}\footnotesize
    \item \emph{Not} a function of $\|v\|_2$ alone $\Rightarrow$ moves the argmin
      even unpenalised. $S_m=m\,\mathrm{CVaR}_{\phi}(|r|)$: penalises fits that
      fail badly on a few rows.
    \item Convex, LP-representable; Hessian $2I$ a.e.\ (drop-in XGB \texttt{obj}).
    \item Cost: re-run everything (CP $\approx494$\,s/cell).
  \end{itemize}
\end{frame}

      \begin{itemize}

          \textbf{Drop robust regression.}

    \item \textbf{CV the synthetic embedded model.} Hard-coded \texttt{rf} now.
    \item \textbf{Fix synthetic scoring.} Oracle is that same \texttt{rf}. Val rows unused

      mixed-type ensemble on noisy \texttt{y\_train}; raise \texttt{n\_kfold}.

      5 radii on 4 d.o.f., no $\delta\in D$ preserves it. Perturb the 4,
      re-derive, re-percentile. In-group $+1$ vs.\ measured $+0.28$.
    \item \textbf{Confirm on the test set.} m-out-of-$n$ at \texttt{ellipsoid},
      real \texttt{X\_test}, GT ensemble. For synthetic, new draws. 
      $\rho^*$ has no training-draw error bars.
    \item \textbf{C-MICL as a fifth method.} Expect infeasible.
    \item \textbf{More baselines.} Feasibility-tuned nominal (either RHS margin
      $f(x) \leq b -m$ or penalty multiplier on outcome's regularization).
    \item Change results to feasibility vs objective, not $\rho$.
    \item \textbf{More instances.} WFP food basket --- Maragno's own wrapper
      setting.
  \end{enumerate}
