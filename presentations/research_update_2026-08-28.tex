\begin{frame}{Method Overview}
    \footnotesize
    \textbf{Problem.} A learned $f(x;\theta)$ is embedded as $f(x)\le b$. Noisy
    labels give models whose ``optimal'' decisions violate the \emph{true} constraint.
    \begin{itemize}
        \item \textbf{Cutting Planes (ours)}: separate the worst-case model over a
              fixed bank of $B{=}200$ relabelings drawn from $D$, one cut per
              iteration. Dial $=\tau$, the stopping tolerance, in sd units.
        \item \textbf{Baselines}: wrapper (chance constraint over $P{=}20$ models,
              dial $\alpha$); RHS margin (shift $b$ by $m\,s_c$);
              C-MICL (split-conformal width model, dial $\alpha$); nominal.
        \item \textbf{Instances}: gastric (OptiCL, 5 toxicity constraints, 96
              held-out arms); DMA-MR reactor (C-MICL Sec.~5.1, \textbf{
              ODE oracle}).
    \end{itemize}
    \textbf{Protocol.}
    \begin{enumerate}
        \item \textbf{Tune} Fix $\rho$; walk each method along \emph{its own} dial;
              report the curve in objective $\times$ feasibility. dial$^\ast$ = best objective
              among cells with feas $\ge0.9$, solved $\ge0.5$. 
        \item \textbf{Evaluate} Re-score at dial$^\ast$ under a judge the dial never faced 
              / on a test set. Different folds of the training set / subsampling are used
              to evaluate the worst-case.
    \end{enumerate}
\end{frame}

\begin{frame}{Implementation details for multiple constraints}
    \footnotesize
    \begin{itemize}
        \item \textbf{Context anchors.} Gastric has one context per study, so
              separation is scored at 10 k-medoid anchors. The protected set is
              \emph{fixed} so all contexts feasible at nominal need to remain
              feasible.
        \item \textbf{Feasibility rollback.} A cut breaking a protected anchor is
              rolled back and that (constraint, draw) pair permanently rejected;
              CP then tries the constraint's \emph{next} most-violating model.
        \item \textbf{Incoherent separation.} Draws are
              ranked \emph{per constraint} and one model admitted for each.
        \item Nothing is ever removed from the master (to prevent cycling).
    \end{itemize}
\end{frame}

\begin{frame}{Gastric: tuned dial, then tested}
    \footnotesize
    Dial$^\ast$ = best objective among cells with feas $\ge 0.9$, solved $\ge 0.5$
    (tuning judge = train-only ensemble). Test judge = full-416 GT ensemble.
    \vspace{0.5em}
    {\scriptsize\setlength{\tabcolsep}{3.5pt}
    \begin{tabular}{llccc|cc|cc}
        \toprule
        & & \multicolumn{3}{c|}{tuned} & \multicolumn{2}{c|}{tested (folds)}
          & \multicolumn{2}{c}{solve time (s)} \\
        method & dial$^\ast$ & feas & obj & solved & feas & obj & master & /arm \\
        \midrule
        nominal              & --             & 0.891 & \textbf{11.30} & 0.951 & 0.934 & \textbf{11.29} & \textbf{0.3} & \textbf{0.08} \\
        RHS margin           & $m=0.2$        & 0.931 & 10.97 & 0.922 & 0.951 & 11.01 & 0.5 & 0.09 \\
        wrapper $\rho{=}1$   & $\alpha=0.4$   & 0.930 & 10.89 & 0.592 & 0.970 & 10.94 & 14.8 & 1.11 \\
        wrapper $\rho{=}0.5$ & $\alpha=0.2$   & 0.941 & 10.88 & 0.679 & 0.961 & 10.97 & 18.3 & 1.36 \\
        CP $\rho{=}0.5$      & $\tau=0.03$    & 0.974 & 10.78 & 0.821 & \textbf{0.987} & 10.86 & 75.8 & 0.20 \\
        CP $\rho{=}1$        & $\tau=0.1$     & 0.938 & 9.84  & 0.870 & 0.947 & 10.07 & 56.7 & 0.22 \\
        C-MICL               & \multicolumn{8}{l}{no cell cleared solved $\ge 0.5$ \quad (master 1.5--8.7)} \\
        \bottomrule
    \end{tabular}}
    \vspace{0.3em}
    \begin{itemize}
        \item \textbf{On one full-data fit nominal already clears 0.9} (0.934).
              The test set is newer studies - less toxicity.
        \item CP ($\rho=0.5$) has best feasibility.
        \item Times are per fold: master $=$ fit${+}$build${+}$solve (for CP the whole
              cut loop); /arm $=$ one prescribe solve per held-out context.
              \textbf{CP pays in the cut loop, not at prescribe time}: $0.20$ vs the
              wrapper's $1.36$\,s/arm --- one cut per iteration against all $P{=}20$ models.
    \end{itemize}
\end{frame}
